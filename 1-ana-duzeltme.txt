alter table public.applications add column if not exists completed_by uuid; -- no FK on purpose: a 2nd applications->profiles FK would make existing API embeds ambiguous

create table if not exists public.job_events (
  id uuid primary key default gen_random_uuid(),
  application_id uuid not null references public.applications(id) on delete cascade,
  listing_id uuid not null references public.listings(id) on delete cascade,
  event text not null check (event in ('completed','completion_reverted')),
  actor_id uuid references public.profiles(id) on delete set null,
  reason text not null default '' check (char_length(reason) <= 500),
  prev_completed_at timestamptz,
  prev_completed_by uuid,
  created_at timestamptz not null default now()
);
create index if not exists job_events_app on public.job_events (application_id, created_at);
alter table public.job_events enable row level security;
revoke all on public.job_events from anon, authenticated;

insert into public.job_events (application_id, listing_id, event, actor_id, reason, created_at)
select a.id, a.listing_id, 'completed', l.owner_id, 'Backfilled by v8 migration from applications.completed_at', a.completed_at
from public.applications a join public.listings l on l.id = a.listing_id
where a.completed_at is not null
  and not exists (select 1 from public.job_events e where e.application_id = a.id and e.event = 'completed');

create or replace function public.job_end_at(d date, st time, et time) returns timestamptz
language sql stable set search_path = public as $$
  select ((d + coalesce(et, time '23:59:59'))
          + case when et is not null and st is not null and et <= st then interval '1 day' else interval '0' end)
         at time zone 'Europe/London'
$$;
grant execute on function public.job_end_at(date, time, time) to anon, authenticated;

create or replace function public.guard_completion_columns() returns trigger
language plpgsql set search_path = public as $$
begin
  if (new.completed_at is distinct from old.completed_at or new.completed_by is distinct from old.completed_by)
     and current_user in ('authenticated', 'anon') then
    raise exception 'Use "Mark job as done" to complete a job';
  end if;
  return new;
end $$;
drop trigger if exists application_completion_guard on public.applications;
create trigger application_completion_guard before update on public.applications
  for each row execute function public.guard_completion_columns();

create or replace function public.check_application_update() returns trigger
language plpgsql security definer set search_path = public as $$
declare owner uuid;
begin
  if auth.uid() is null then return new; end if; -- dashboard / admin edits
  select owner_id into owner from listings where id = old.listing_id;
  if new.listing_id <> old.listing_id or new.applicant_id <> old.applicant_id or new.message <> old.message then
    raise exception 'Only the status can change';
  end if;
  if is_admin() and new.status = old.status
     and new.price is not distinct from old.price and new.price_unit is not distinct from old.price_unit then
    return new;
  end if;
  if (new.price is distinct from old.price or new.price_unit is distinct from old.price_unit)
     and not (auth.uid() = old.applicant_id and new.status = 'pending') then
    raise exception 'Only the applicant can change their quote';
  end if;
  if auth.uid() = old.applicant_id then
    if old.status = 'invited' and new.status in ('accepted','declined') then return new; end if;
    if old.status in ('pending','declined','withdrawn') and new.status in ('withdrawn','pending') then return new; end if;
  end if;
  if auth.uid() = owner and old.status = 'pending' and new.status = 'accepted'
     and old.price is not null and coalesce(old.quoted_at, old.created_at) < now() - interval '48 hours' then
    raise exception 'This quote has expired. Ask for an updated quote first.';
  end if;
  if auth.uid() = owner and old.status <> 'invited' and new.status in ('accepted','declined','pending') then return new; end if;
  if auth.uid() = owner and old.status = 'invited' and new.status = 'withdrawn' then return new; end if;
  raise exception 'Not allowed';
end $$;

create or replace function public.complete_job(a uuid) returns void
language plpgsql security definer set search_path = public as $$
declare r record; ends timestamptz;
begin
  if auth.uid() is null then raise exception 'Not signed in'; end if;
  select ap.id, ap.applicant_id, ap.status, ap.completed_at, ap.listing_id,
         l.owner_id, l.title, l.status as lstatus, l.date, l.start_time, l.end_time into r
  from applications ap join listings l on l.id = ap.listing_id where ap.id = a
  for update of ap;
  if r.id is null or r.owner_id <> auth.uid() then raise exception 'Not allowed'; end if;
  if is_banned() then raise exception 'Not allowed'; end if;
  if r.status <> 'accepted' then raise exception 'Only accepted jobs can be completed'; end if;
  if r.lstatus = 'closed' then raise exception 'This listing was closed'; end if;
  if r.completed_at is not null then raise exception 'Already completed'; end if;
  ends := job_end_at(r.date, r.start_time, r.end_time);
  if now() < ends then
    raise exception 'NOT_FINISHED: This job can be marked as done after %', to_char(ends at time zone 'Europe/London', 'DD Mon YYYY HH24:MI');
  end if;
  update applications set completed_at = now(), completed_by = auth.uid() where id = a;
  insert into job_events (application_id, listing_id, event, actor_id) values (a, r.listing_id, 'completed', auth.uid());
  perform notify(r.applicant_id, 'completed', 'Job done: ' || r.title, 'Leave a review to help others choose.', '/chat/' || a);
  perform notify(r.owner_id, 'completed', 'How did it go? ' || r.title, 'Leave a review to help others choose.', '/chat/' || a);
end $$;
revoke execute on function public.complete_job(uuid) from public, anon;
grant execute on function public.complete_job(uuid) to authenticated;

create or replace function public.check_review() returns trigger
language plpgsql security definer set search_path = public as $$
declare ap record;
begin
  select a.applicant_id, a.status, a.completed_at, l.owner_id into ap
  from applications a join listings l on l.id = a.listing_id where a.id = new.application_id;
  if ap is null then raise exception 'Job not found'; end if;
  if not ((new.reviewer_id = ap.applicant_id and new.reviewee_id = ap.owner_id)
       or (new.reviewer_id = ap.owner_id and new.reviewee_id = ap.applicant_id)) then
    raise exception 'Not a participant';
  end if;
  if ap.status <> 'accepted' or ap.completed_at is null then
    raise exception 'NOT_COMPLETED: You can rate this job once it has been completed';
  end if;
  if exists (select 1 from reviews where application_id = new.application_id and reviewer_id = new.reviewer_id) then
    raise exception 'ALREADY_REVIEWED: You have already rated this job';
  end if;
  new.created_at := now(); -- server timestamp, not client supplied
  return new;
end $$;
drop trigger if exists review_guard on public.reviews;
create trigger review_guard before insert on public.reviews
  for each row execute function public.check_review();

create or replace function public.admin_recent_completions() returns json
language plpgsql stable security definer set search_path = public as $$
begin
  if not is_admin() then raise exception 'Not allowed'; end if;
  return coalesce((select json_agg(x order by x.completed_at desc) from (
    select a.id, a.completed_at, a.completed_by, cb.name as completed_by_name,
      l.id as listing_id, l.title, l.category, l.date, l.start_time, l.end_time,
      job_end_at(l.date, l.start_time, l.end_time) as ends_at,
      a.completed_at < job_end_at(l.date, l.start_time, l.end_time) as early,
      po.name as owner_name, pa.name as applicant_name,
      (select count(*) from reviews r where r.application_id = a.id) as reviews
    from applications a join listings l on l.id = a.listing_id
    left join profiles cb on cb.id = a.completed_by
    left join profiles po on po.id = l.owner_id
    left join profiles pa on pa.id = a.applicant_id
    where a.completed_at is not null
    order by a.completed_at desc limit 100) x), '[]'::json);
end $$;

create or replace function public.admin_job_history(a uuid) returns json
language plpgsql stable security definer set search_path = public as $$
begin
  if not is_admin() then raise exception 'Not allowed'; end if;
  return (select json_build_object(
    'id', ap.id, 'status', ap.status, 'created_at', ap.created_at,
    'completed_at', ap.completed_at, 'completed_by', ap.completed_by,
    'listing_id', l.id, 'title', l.title, 'date', l.date, 'start_time', l.start_time, 'end_time', l.end_time,
    'ends_at', job_end_at(l.date, l.start_time, l.end_time),
    'owner_name', po.name, 'applicant_name', pa.name,
    'events', coalesce((select json_agg(json_build_object(
        'id', e.id, 'event', e.event, 'actor_id', e.actor_id, 'actor_name', pe.name, 'reason', e.reason,
        'prev_completed_at', e.prev_completed_at, 'created_at', e.created_at) order by e.created_at)
      from job_events e left join profiles pe on pe.id = e.actor_id where e.application_id = ap.id), '[]'::json))
    from applications ap join listings l on l.id = ap.listing_id
    left join profiles po on po.id = l.owner_id
    left join profiles pa on pa.id = ap.applicant_id
    where ap.id = a);
end $$;

create or replace function public.admin_revert_completion(a uuid, reason text) returns void
language plpgsql security definer set search_path = public as $$
declare r record;
begin
  if not is_admin() then raise exception 'Not allowed'; end if;
  if char_length(coalesce(trim(reason), '')) not between 5 and 500 then raise exception 'Please give a reason (5-500 characters)'; end if;
  select ap.id, ap.listing_id, ap.completed_at, ap.completed_by into r from applications ap where ap.id = a for update;
  if r.id is null then raise exception 'Job not found'; end if;
  if r.completed_at is null then raise exception 'This job is not marked as completed'; end if;
  insert into job_events (application_id, listing_id, event, actor_id, reason, prev_completed_at, prev_completed_by)
  values (a, r.listing_id, 'completion_reverted', auth.uid(), trim(reason), r.completed_at, r.completed_by);
  update applications set completed_at = null, completed_by = null where id = a;
end $$;

revoke execute on function public.admin_recent_completions() from public, anon;
revoke execute on function public.admin_job_history(uuid) from public, anon;
revoke execute on function public.admin_revert_completion(uuid, text) from public, anon;
grant execute on function public.admin_recent_completions() to authenticated;
grant execute on function public.admin_job_history(uuid) to authenticated;
grant execute on function public.admin_revert_completion(uuid, text) to authenticated;
